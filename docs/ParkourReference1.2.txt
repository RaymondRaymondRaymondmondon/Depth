3.1 FABRIK Implementation with Joint Angle Constraints 
C++ 
struct BoneJoint { 
    Vector2D position; 
    float length; 
    float minAngle; // Relative to parent joint 
    float maxAngle; 
}; 
class ConstrainedFABRIKSolver { 
public: 
    static void SolveChain(std::vector<BoneJoint>& chain, Vector2D target, float tolerance) { 
        size_t n = chain.size(); 
        if (n < 2) return; 
        // 1. Check reachability 
        float totalLength = 0.0f; 
        for (size_t i = 0; i < n - 1; ++i) totalLength += chain[i].length; 
        Vector2D root = chain[0].position; 
        if (Distance(root, target) >= totalLength) { 
            for (size_t i = 0; i < n - 1; ++i) { 
                Vector2D dir = Normalize(target - chain[i].position); 
                chain[i + 1].position = chain[i].position + dir * chain[i].length; 
            } 
            return; 
        } 
        // 2. Iterative FABRIK Pass 
        int iterations = 0; 
        while (Distance(chain[n - 1].position, target) > tolerance && iterations < 8) { 
            // Backward Pass 
            chain[n - 1].position = target; 
            for (int i = static_cast<int>(n) - 2; i >= 0; --i) { 
                Vector2D dir = Normalize(chain[i].position - chain[i + 1].position); 
                chain[i].position = chain[i + 1].position + dir * chain[i].length; 
            } 
            // Forward Pass with Angular Constraints 
            chain[0].position = root; 
            for (size_t i = 0; i < n - 1; ++i) { 
                Vector2D dir = Normalize(chain[i + 1].position - chain[i].position); 
                // Apply Angular Constraint relative to parent segment angle 
                if (i > 0) { 
                    Vector2D parentDir = Normalize(chain[i].position - chain[i - 1].position); 
                    dir = ConstrainAngle(parentDir, dir, chain[i].minAngle, chain[i].maxAngle); 
                } 
                chain[i + 1].position = chain[i].position + dir * chain[i].length; 
            } 
            iterations++; 
        } 
    } 
private: 
    static Vector2D ConstrainAngle(Vector2D parentDir, Vector2D currentDir, float minA, float maxA) { 
        float parentAngle = atan2(parentDir.y, parentDir.x); 
        float currentAngle = atan2(currentDir.y, currentDir.x); 
        float diff = currentAngle - parentAngle; 
        // Wrap to [-PI, PI] 
        while (diff < -3.14159f) diff += 6.28318f; 
        while (diff > 3.14159f) diff -= 6.28318f; 
        // Clamp angle 
        float clampedDiff = std::clamp(diff, minA, maxA); 
        float finalAngle = parentAngle + clampedDiff; 
        return { cos(finalAngle), sin(finalAngle) }; 
    } 
};

struct PersonalityProfile {

class UtilityEngine { 
public: 
    static float CalculateHuntUtility(const Entity& self, const Entity& target, const PersonalityProfile& p) { 
        float hunger = self.hungerLevel; // [0.0 = Satiated, 1.0 = Starving] 
        float sizeRatio = target.mass / self.mass; 
        float distance = Distance(self.position, target.position); 
        if (sizeRatio > 1.2f && p.bravery < 0.6f) return 0.0f; // Won't attack larger prey unless brave 
        float distanceScore = 1.0f - (distance / self.perceptionRadius); 
        float baseUtility = (hunger * 0.5f) + (distanceScore * 0.3f) + (p.aggression * 0.2f); 
        // Apply Abnormal Overrides 
        switch (p.overrideType) { 
            case AbnormalPersonality::RabidEnraged: 
                return 1.0f; // Always max hunt utility regardless of size/hunger 
            case AbnormalPersonality::GluttonousDevourer: 
                return baseUtility * 2.5f; 
            case AbnormalPersonality::Kleptomaniac: 
                return 0.0f; // Replaces hunting with item stealing 
            default: 
                return std::clamp(baseUtility, 0.0f, 1.0f); 
        } 
    } 
};

class BehaviorStateMachine { 
public: 
    static void UpdateEntityBehavior(Entity& entity, EnvironmentContext& ctx) { 
        // 1. Process Senses 
        PerceptionData perception = SensorySystem::Process(entity, ctx); 
        // 2. Evaluate Abnormal Overrides 
        if (entity.personality.overrideType != AbnormalPersonality::None) { 
            ExecuteAbnormalBehavior(entity, perception, ctx); 
            return; 
        } 
        // 3. Standard Utility Evaluation 
        float uHunt = UtilityEngine::CalculateHuntUtility(entity, perception.bestPrey, entity.personality); 
        float uFlee = UtilityEngine::CalculateFleeUtility(entity, perception.biggestThreat, entity.personality); 
        float uInvestigate = UtilityEngine::CalculateInvestigateUtility(entity, perception.loudestSound, entity.personality); 
        // 4. Select Dominant Utility Action 
        if (uFlee > uHunt && uFlee > 0.5f) { 
            entity.SetState(ActionState::Fleeing, perception.biggestThreat.position); 
        } else if (uHunt > 0.4f) { 
            entity.SetState(ActionState::Hunting, perception.bestPrey.position); 
        } else if (uInvestigate > 0.3f) { 
            entity.SetState(ActionState::Investigating, perception.loudestSound.position); 
        } else { 
            entity.SetState(ActionState::Wandering); 
        } 
    } 
private: 
    static void ExecuteAbnormalBehavior(Entity& e, const PerceptionData& p, EnvironmentContext& ctx) { 
        switch (e.personality.overrideType) { 
            case AbnormalPersonality::RabidEnraged: 
                // Attacks nearest target regardless of species, suffers continuous health loss 
                e.health -= ctx.deltaTime * 1.5f; 
                e.targetPos = p.nearestLivingEntity.position; 
                e.SetState(ActionState::Attacking); 
                break; 
            case AbnormalPersonality::Kleptomaniac: 
                // Tracks player's loose relics or inventory items, grabs, and flees to level boundary 
                if (ctx.playerHasRelic) { 
                    e.targetPos = ctx.playerPosition; 
                    if (Distance(e.position, ctx.playerPosition) < 1.0f) { 
                        ctx.StealRelic(); 
                        e.personality.overrideType = AbnormalPersonality::None; // Shift to standard flee 
                        e.SetState(ActionState::Fleeing); 
                    } 
                } 
                break; 
            case AbnormalPersonality::HyperProtectiveParent: 
                // Peaceful until player/predator approaches home Den (< 6.0m), then acquires hyper-speed 
                if (Distance(ctx.playerPosition, e.homeDenPosition) < 6.0f) { 
                    e.movementForceMultiplier = 2.2f; 
                    e.targetPos = ctx.playerPosition; 
                    e.SetState(ActionState::Attacking); 
                } else { 
                    e.targetPos = e.homeDenPosition; 
                    e.SetState(ActionState::Guarding); 
                } 
                break; 
            case AbnormalPersonality::SuicidalSelfDestructive: 
                // Seeks explosive triggers (gunpowder barrels, electrical arcs) near targets 
                if (ctx.HasNearbyExplosive()) { 
                    e.targetPos = ctx.GetNearestExplosivePos(); 
                    if (Distance(e.position, e.targetPos) < 0.8f) { 
                        ctx.TriggerExplosion(e.position); 
                    } 
                } 
                break; 
            case AbnormalPersonality::ParasiticHost: 
                // Latches onto larger entities or player, injecting control vectors into their inputs 
                if (!e.isLatched) { 
                    e.targetPos = p.biggestThreat.position; 
                    if (Distance(e.position, e.targetPos) < 0.5f) { 
                        e.LatchToTarget(p.biggestThreat); 
                    } 
                } 
                break; 
            case AbnormalPersonality::CovetousHoarder: 
                // Collects loose items and drags them to its nest 
                if (!e.isCarryingItem) { 
                    e.targetPos = ctx.GetNearestLooseItemPos(); 
                } else { 
                    e.targetPos = e.homeDenPosition; 
                } 
                break; 
            case AbnormalPersonality::SymbioticCompanion: 
                // Emits light/speed boost aura and follows player closely 
                e.targetPos = ctx.playerPosition + Vector2D{-1.5f, -1.0f}; 
                ctx.ApplySpeedAura(e.position, 4.0f); 
                break; 
            case AbnormalPersonality::TerritorialTyrant: 
                // Dominates a 12m radius zone. Attacks ANY entity entering 
                for (auto& target : ctx.GetEntitiesInRadius(e.territoryCenter, 12.0f)) { 
                    if (target.id != e.id) { 
                        e.targetPos = target.position; 
                        e.SetState(ActionState::Attacking); 
                        return; 
                    } 
                } 
                break; 
            case AbnormalPersonality::PyromaniacShockSeeker: 
                // Drawn directly to electrical arcs, light pulses, or powder trails 
                e.targetPos = ctx.GetLoudestEnergySourcePos(); 
                break; 
            case AbnormalPersonality::GluttonousDevourer: 
                // Consumes anything smaller than itself, gaining 5% mass per consumption 
                if (p.bestPrey.mass < e.mass) { 
                    e.targetPos = p.bestPrey.position; 
                    if (Distance(e.position, e.targetPos) < 0.8f) { 
                        e.Consume(p.bestPrey); 
                        e.mass *= 1.05f; 
                        e.scale *= 1.03f; 
                    } 
                } 
                break; 
            case AbnormalPersonality::PhobicNyctophobic: 
                // Panics when ambient light < 0.3, fleeing toward nearest bioluminescent source 
                if (ctx.GetLightLevel(e.position) < 0.3f) { 
                    e.targetPos = ctx.GetNearestLightSourcePos(); 
                    e.movementForceMultiplier = 1.8f; 
                    e.SetState(ActionState::Fleeing); 
                } 
                break; 
            case AbnormalPersonality::MalignantAlpha: 
                // Emits continuous command wave, forcing all same-species within 15m to join its target attack 
                e.targetPos = ctx.playerPosition; 
                ctx.ForceSwarmTarget(e.speciesID, e.position, 15.0f, ctx.playerPosition); 
                break; 
        } 
    }

5.1 Auto-Tiling System (Dual-Grid / Bitmask 8-Bit Rule) 
Level terrain surfaces are rendered by evaluating surrounding cell states using an 8-bit bitmask rule to select correct decorative edge tiles (cracked metal, barnacle crusts, overgrown moss, carved Atlantean marble). 
C++ 
uint8_t ComputeTileBitmask(const Grid& grid, int x, int y) { 
    uint8_t mask = 0; 
    if (grid.IsSolid(x,     y - 1)) mask |= 1;   // North 
    if (grid.IsSolid(x + 1, y - 1)) mask |= 2;   // North-East 
    if (grid.IsSolid(x + 1, y    )) mask |= 4;   // East 
    if (grid.IsSolid(x + 1, y + 1)) mask |= 8;   // South-East 
    if (grid.IsSolid(x    , y + 1)) mask |= 16;  // South 
    if (grid.IsSolid(x - 1, y + 1)) mask |= 32;  // South-West 
    if (grid.IsSolid(x - 1, y    )) mask |= 64;  // West 
    if (grid.IsSolid(x - 1, y - 1)) mask |= 128; // North-West 
    return mask; 
}

    vec2 coord = FragTexCoord; 
    // Calculate sine-wave offset grid for underwater distortion

struct BoneJoint { 
    Vector2D position; 
    float length; 
    float minAngle; // Relative to parent joint 
    float maxAngle; 
}; 
class ConstrainedFABRIKSolver { 
public: 
    static void SolveChain(std::vector<BoneJoint>& chain, Vector2D target, float tolerance) { 
        size_t n = chain.size(); 
        if (n < 2) return; 
        // 1. Check reachability 
        float totalLength = 0.0f; 
        for (size_t i = 0; i < n - 1; ++i) totalLength += chain[i].length; 
        Vector2D root = chain[0].position; 
        if (Distance(root, target) >= totalLength) { 
            for (size_t i = 0; i < n - 1; ++i) { 
                Vector2D dir = Normalize(target - chain[i].position); 
                chain[i + 1].position = chain[i].position + dir * chain[i].length; 
            } 
            return; 
        } 
        // 2. Iterative FABRIK Pass 
        int iterations = 0; 
        while (Distance(chain[n - 1].position, target) > tolerance && iterations < 8) { 
            // Backward Pass 
            chain[n - 1].position = target; 
            for (int i = static_cast<int>(n) - 2; i >= 0; --i) { 
                Vector2D dir = Normalize(chain[i].position - chain[i + 1].position); 
                chain[i].position = chain[i + 1].position + dir * chain[i].length; 
            } 
            // Forward Pass with Angular Constraints 
            chain[0].position = root; 
            for (size_t i = 0; i < n - 1; ++i) { 
                Vector2D dir = Normalize(chain[i + 1].position - chain[i].position); 
                // Apply Angular Constraint relative to parent segment angle 
                if (i > 0) { 
                    Vector2D parentDir = Normalize(chain[i].position - chain[i - 1].position); 
                    dir = ConstrainAngle(parentDir, dir, chain[i].minAngle, chain[i].maxAngle); 
                } 
                chain[i + 1].position = chain[i].position + dir * chain[i].length; 
            } 
            iterations++; 
        } 
    } 
private: 
    static Vector2D ConstrainAngle(Vector2D parentDir, Vector2D currentDir, float minA, float maxA) { 
        float parentAngle = atan2(parentDir.y, parentDir.x); 
        float currentAngle = atan2(currentDir.y, currentDir.x); 
        float diff = currentAngle - parentAngle; 
        // Wrap to [-PI, PI] 
        while (diff < -3.14159f) diff += 6.28318f; 
        while (diff > 3.14159f) diff -= 6.28318f; 
        // Clamp angle 
        float clampedDiff = std::clamp(diff, minA, maxA); 
        float finalAngle = parentAngle + clampedDiff; 
        return { cos(finalAngle), sin(finalAngle) }; 
    } 
};

        // 4. Select Dominant Utility Action

            entity.SetState(ActionState::Hunting, perception.bestPrey.position); 
        } else if (uInvestigate > 0.3f) { 
            entity.SetState(ActionState::Investigating, perception.loudestSound.position);

            entity.SetState(ActionState::Wandering);

                e.health -= ctx.deltaTime * 1.5f; 
                e.targetPos = p.nearestLivingEntity.position;

                    if (Distance(e.position, ctx.playerPosition) < 1.0f) {

                        e.SetState(ActionState::Fleeing);

                    e.movementForceMultiplier = 2.2f; 
                    e.targetPos = ctx.playerPosition;

                    e.SetState(ActionState::Guarding);

                }

            case AbnormalPersonality::CovetousHoarder: 
                // Collects loose items and drags them to its nest 
                if (!e.isCarryingItem) { 
                    e.targetPos = ctx.GetNearestLooseItemPos(); 
                } else { 
                    e.targetPos = e.homeDenPosition; 
                } 
                break; 
            case AbnormalPersonality::SymbioticCompanion:

5.1 Auto-Tiling System (Dual-Grid / Bitmask 8-Bit Rule) 
Level terrain surfaces are rendered by evaluating surrounding cell states using an 8-bit bitmask rule to select correct decorative edge tiles (cracked metal, barnacle crusts, overgrown moss, carved Atlantean marble). 
uint8_t ComputeTileBitmask(const Grid& grid, int x, int y) { 
    uint8_t mask = 0; 
    if (grid.IsSolid(x,     y - 1)) mask |= 1;   // North 
    if (grid.IsSolid(x + 1, y - 1)) mask |= 2;   // North-East 
    if (grid.IsSolid(x + 1, y    )) mask |= 4;   // East 
    if (grid.IsSolid(x + 1, y + 1)) mask |= 8;   // South-East 
    if (grid.IsSolid(x    , y + 1)) mask |= 16;  // South 
    if (grid.IsSolid(x - 1, y + 1)) mask |= 32;  // South-West 
    if (grid.IsSolid(x - 1, y    )) mask |= 64;  // West 
    if (grid.IsSolid(x - 1, y - 1)) mask |= 128; // North-West 
    return mask; 
}

enum class TraversalState { Sprint, Slide, ImpactRoll, HydroGlide, WaterDash, ParachuteBrake }; 
struct MaterialProperties { 
    float friction = 0.8f; 
    float elasticity = 0.2f; 
    float brakeThreshold = 500.0f; // Force required to break material 
    bool generatesSpores = false; 
}; 
struct PhysicsBody { 
    sf::Vector2f position{0.0f, 0.0f}; 
    sf::Vector2f velocity{0.0f, 0.0f}; 
    sf::Vector2f acceleration{0.0f, 0.0f}; 
    float mass = 70.0f;           // kg 
    float volume = 0.07f;         // m^3 
    float dragCoefficient = 0.1f; 
    float crossSectionArea = 0.5f; // m^2 
    bool isSubmerged = false; 
    TraversalState state = TraversalState::Sprint; 
}; 
class PhysicsIntegrator { 
public: 
    static void Integrate(PhysicsBody& body, const sf::Vector2f& fluidVelocity, float fluidDensity, float dt) { 
        // 1. Calculate Relative Velocity 
        sf::Vector2f relVel = body.velocity - fluidVelocity; 
        float speed = std::sqrt(relVel.x * relVel.x + relVel.y * relVel.y); 
        // 2. Gravitational & Buoyant Forces 
        float g = 9.81f * 100.0f; // Scale factor: 1 meter = 100 pixels 
        sf::Vector2f fGravity = {0.0f, body.mass * g}; 
        sf::Vector2f fBuoyancy = {0.0f, 0.0f}; 
        if (body.isSubmerged) { 
            fBuoyancy.y = -(fluidDensity * body.volume * g); 
        } 
        // 3. Hydrodynamic Drag 
        sf::Vector2f fDrag = {0.0f, 0.0f}; 
        if (speed > 0.0001f) { 
            float dragMag = 0.5f * fluidDensity * body.dragCoefficient * body.crossSectionArea * speed * speed; 
            fDrag = -(relVel / speed) * dragMag; 
        } 
        // 4. Sum Accelerations 
        sf::Vector2f netForce = fGravity + fBuoyancy + fDrag; 
        sf::Vector2f newAccel = netForce / body.mass; 
        // 5. Velocity-Verlet Integration 
        body.position += body.velocity * dt + 0.5f * body.acceleration * (dt * dt); 
        body.velocity += 0.5f * (body.acceleration + newAccel) * dt; 
        body.acceleration = newAccel; 
    } 
};

3.1 C++ Scent Grid Implementation (1;=FL%JA<}@HH)  
#pragma once 
#include <vector> 
#include <cmath> 
#include <SFML/System/Vector2.hpp> 
class ScentGrid { 
private: 
    int width, height; 
    float cellSize; 
    float diffusionRate = 0.15f; 
    float decayRate = 0.02f; 
    std::vector<float> grid; 
    std::vector<float> nextGrid; 
public: 
    ScentGrid(int w, int h, float scale) : width(w), height(h), cellSize(scale) { 
        grid.resize(w * h, 0.0f); 
        nextGrid.resize(w * h, 0.0f); 
    } 
    void EmitScent(int x, int y, float amount) { 
        if (x >= 0 && x < width && y >= 0 && y < height) { 
            grid[y * width + x] += amount; 
        } 
    } 
    void Update(float dt, const std::vector<sf::Vector2f>& fluidVelocities) { 
        for (int y = 1; y < height - 1; ++y) { 
            for (int x = 1; x < width - 1; ++x) { 
                int idx = y * width + x; 
                // Discrete Laplace Operator (Diffusion) 
                float laplacian = grid[idx - 1] + grid[idx + 1] + 
                                  grid[(y - 1) * width + x] + grid[(y + 1) * width + x] - 4.0f * grid[idx]; 
                // Fluid Advection Offset 
                sf::Vector2f vel = fluidVelocities[idx]; 
                int advX = std::clamp(x - (int)(vel.x * dt / cellSize), 0, width - 1); 
                int advY = std::clamp(y - (int)(vel.y * dt / cellSize), 0, height - 1); 
                float advectedScent = grid[advY * width + advX]; 
                // Diffusion + Advection + Decay 
                nextGrid[idx] = advectedScent + (diffusionRate * laplacian - decayRate * advectedScent) * dt; 
                nextGrid[idx] = std::max(0.0f, nextGrid[idx]); 
            } 
        } 
        grid = nextGrid; 
    } 
    sf::Vector2f SampleGradient(float worldX, float worldY) const { 
        int gx = std::clamp((int)(worldX / cellSize), 1, width - 2); 
        int gy = std::clamp((int)(worldY / cellSize), 1, height - 2); 
        float dx = (grid[gy * width + (gx + 1)] - grid[gy * width + (gx - 1)]) * 0.5f; 
        float dy = (grid[(gy + 1) * width + gx] - grid[(gy - 1) * width + gx]) * 0.5f; 
        return {dx, dy}; 
    } 
};

uint8_t ComputeTileBitmask(const std::vector<uint8_t>& grid, int x, int y, int gridWidth) { 
    auto IsSolid = [&](int cx, int cy) -> uint8_t { 
        if (cx < 0 || cx >= gridWidth || cy < 0) return 1; 
        return grid[cy * gridWidth + cx] > 0 ? 1 : 0; 
    }; 
    uint8_t mask = 0; 
    if (IsSolid(x,     y - 1)) mask |= 1;   // North 
    if (IsSolid(x + 1, y - 1)) mask |= 2;   // North-East 
    if (IsSolid(x + 1, y    )) mask |= 4;   // East 
    if (IsSolid(x + 1, y + 1)) mask |= 8;   // South-East 
    if (IsSolid(x    , y + 1)) mask |= 16;  // South 
    if (IsSolid(x - 1, y + 1)) mask |= 32;  // South-West 
    if (IsSolid(x - 1, y    )) mask |= 64;  // West 
    if (IsSolid(x - 1, y - 1)) mask |= 128; // North-West 
    return mask; 
}

        sf::Vector2f targetForward,

        float dt 
    ) {

                break;

            case MermanRole::AlphaLeader: {

                desiredPosition = targetPos - targetForward * (targetDistanceOffset * 1.5f); 
                break;

#pragma once 
#include <SFML/System/Vector2.hpp> 
#include <cmath> 
#include <algorithm> 
class TigerSharkAIController { 
public: 
    enum class SharkState { Patrol, ScentTracking, TestingNudge, BloodFrenzy, Retreating }; 
    SharkState state = SharkState::Patrol; 
    float spiralAngle = 0.0f; 
    float frenzyTimer = 0.0f; 
    float health = 250.0f; 
    float maxHealth = 250.0f; 
    sf::Vector2f UpdateSharkBehavior( 
        sf::Vector2f sharkPos,  
        sf::Vector2f targetPos,  
        float localBloodScent,  
        sf::Vector2f scentGradient,  
        float dt 
    ) { 
        // 1. Evaluate State Transitions based on Scent and Damage 
        if (health < maxHealth * 0.25f && state != SharkState::BloodFrenzy) { 
            state = SharkState::Retreating; // Low health escape unless in frenzy 
        } else if (localBloodScent > 0.40f || (state == SharkState::TestingNudge && localBloodScent > 0.15f)) { 
            state = SharkState::BloodFrenzy; 
            frenzyTimer = 8.0f; // 8 seconds of relentless attack 
        } else if (localBloodScent > 0.05f && state == SharkState::Patrol) { 
            state = SharkState::ScentTracking; 
        } 
        // 2. State Actions 
        sf::Vector2f toTarget = targetPos - sharkPos; 
        float dist = std::sqrt(toTarget.x * toTarget.x + toTarget.y * toTarget.y); 
        switch (state) { 
            case SharkState::Patrol: { 
                // Wide, smooth sinusoidal swimming trajectory 
                spiralAngle += dt * 0.5f; 
                return sf::Vector2f{std::cos(spiralAngle), std::sin(spiralAngle)} * 80.0f; 
            } 
            case SharkState::ScentTracking: { 
                // Follows scent gradient vector, transition to logarithmic spiral near target 
                if (dist > 6.0f) { 
                    float gradMag = std::sqrt(scentGradient.x * scentGradient.x + scentGradient.y * scentGradient.y); 
                    return (gradMag > 0.001f) ? (scentGradient / gradMag) * 180.0f : toTarget / dist * 180.0f; 
                } else { 
                    // Logarithmic Spiral trajectory computation 
                    spiralAngle += dt * 1.8f; 
                    float radius = 6.0f * std::exp(-0.15f * spiralAngle); 
                    sf::Vector2f spiralOffset{std::cos(spiralAngle) * radius, std::sin(spiralAngle) * radius}; 
                    sf::Vector2f desiredPos = targetPos + spiralOffset; 
                    if (dist < 1.5f) state = SharkState::TestingNudge; 
                    return (desiredPos - sharkPos) * 2.5f; 
                } 
            } 
            case SharkState::TestingNudge: { 
                // Fast direct bump to gauge prey response/mass 
                return (toTarget / dist) * 350.0f; 
            } 
            case SharkState::BloodFrenzy: { 
                // Maximum speed, direct trajectory, ignores turn damping 
                frenzyTimer -= dt; 
                if (frenzyTimer <= 0.0f) state = SharkState::ScentTracking; 
                return (toTarget / dist) * 550.0f; // Extreme frenzy speed 
            } 
            case SharkState::Retreating: { 
                return -(toTarget / dist) * 300.0f; 
            } 
        } 
        return {0.0f, 0.0f}; 
    } 
};

public: 
    std::vector<BatBoid> bats; 
    float separationRadius = 0.6f; 
    float neighborRadius = 3.0f; 
    void UpdateSwarm(sf::Vector2f soundOrigin, float soundIntensity, float dt) { 
        for (auto& bat : bats) { 
            sf::Vector2f separation{0.0f, 0.0f}; 
            sf::Vector2f alignment{0.0f, 0.0f}; 
            sf::Vector2f cohesion{0.0f, 0.0f}; 
            int neighborCount = 0; 
            for (const auto& other : bats) { 
                if (&bat == &other) continue; 
                sf::Vector2f diff = bat.position - other.position; 
                float dist = std::sqrt(diff.x * diff.x + diff.y * diff.y); 
                if (dist < separationRadius && dist > 0.001f) { 
                    separation += (diff / dist) / dist; // Quadratic repulsion 
                } 
                if (dist < neighborRadius) { 
                    alignment += other.velocity; 
                    cohesion += other.position; 
                    neighborCount++; 
                } 
            } 
            if (neighborCount > 0) { 
                alignment /= static_cast<float>(neighborCount); 
                cohesion = ((cohesion / static_cast<float>(neighborCount)) - bat.position) * 0.5f; 
            } 
            // Sound Source Attractor Force 
            sf::Vector2f toSound = soundOrigin - bat.position; 
            float soundDist = std::sqrt(toSound.x * toSound.x + toSound.y * toSound.y); 
            sf::Vector2f soundAttractor = (soundDist > 0.001f) ? (toSound / soundDist) * (soundIntensity * 2.0f) : sf::Vector2f{0.0f, 0.0f}; 
            // Calculate Net Acceleration Vector 
            sf::Vector2f accel = separation * 2.5f + alignment * 1.0f + cohesion * 1.0f + soundAttractor * 1.8f; 
            bat.velocity += accel * dt; 
            // Clamp Max Speed 
            float speed = std::sqrt(bat.velocity.x * bat.velocity.x + bat.velocity.y * bat.velocity.y); 
            if (speed > 350.0f) bat.velocity = (bat.velocity / speed) * 350.0f; 
            bat.position += bat.velocity * dt; 
        } 
    } 
};
